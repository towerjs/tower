This tutorial walks through building a todo application with Tower.

## 1. Create a Migration

```ts filename="src/vault/migrations/0001_todos.ts"
import { type Vault } from '@towerjs/vault'

export async function up(db: Vault) {
  await db.schema
    .createTable('todos')
    .addColumn('id', 'uuid', (col) => col.primaryKey().defaultTo(db.fn('gen_random_uuid')))
    .addColumn('title', 'varchar(255)', (col) => col.notNull())
    .addColumn('completed', 'boolean', (col) => col.notNull().defaultTo(false))
    .addColumn('user_id', 'uuid', (col) => col.notNull())
    .addColumn('created_at', 'timestamptz', (col) => col.defaultTo(db.fn('now')))
    .addColumn('updated_at', 'timestamptz', (col) => col.defaultTo(db.fn('now')))
    .execute()
}

export async function down(db: Vault) {
  await db.schema.dropTable('todos').execute()
}
```

## 2. Run Migrations

```bash
npx tower migrate
```

## 3. Server Actions

```ts filename="src/actions/todos.ts"
'use server'

import { gatehouse } from '@towerjs/tower/gatehouse'
import { vault } from '@towerjs/tower/vault'

export async function addTodo(formData: FormData) {
  const session = await gatehouse.getSession()
  if (!session) throw new Error('Not authenticated')

  const title = formData.get('title') as string
  await vault.insertInto('todos').values({ title, user_id: session.user.id }).execute()
}

export async function toggleTodo(formData: FormData) {
  const id = formData.get('id') as string
  const todo = await vault.selectFrom('todos').select('completed').where('id', '=', id).executeTakeFirst()

  await vault.updateTable('todos').set({ completed: !todo?.completed }).where('id', '=', id).execute()
}

export async function deleteTodo(formData: FormData) {
  const id = formData.get('id') as string
  await vault.deleteFrom('todos').where('id', '=', id).execute()
}
```

## 4. Todo Page

```tsx filename="src/app/todos/page.tsx"
import { addTodo, deleteTodo, toggleTodo } from '@/actions/todos'

import { gatehouse } from '@towerjs/tower/gatehouse'
import { vault } from '@towerjs/tower/vault'

export default async function TodosPage() {
  const session = await gatehouse.requireUser()
  const todos = await vault
    .selectFrom('todos')
    .selectAll()
    .where('user_id', '=', session.id)
    .orderBy('created_at', 'desc')
    .execute()

  return (
    <div>
      <h1>My Todos</h1>

      <form action={addTodo}>
        <input name="title" placeholder="New todo..." required />
        <button type="submit">Add</button>
      </form>

      <ul>
        {todos.map((todo) => (
          <li key={todo.id}>
            <form action={toggleTodo} style={{ display: 'inline' }}>
              <input type="hidden" name="id" value={todo.id} />
              <button type="submit">{todo.completed ? '☑' : '○'}</button>
            </form>
            <span style={{ textDecoration: todo.completed ? 'line-through' : 'none' }}>{todo.title}</span>
            <form action={deleteTodo} style={{ display: 'inline' }}>
              <input type="hidden" name="id" value={todo.id} />
              <button type="submit">✕</button>
            </form>
          </li>
        ))}
      </ul>
    </div>
  )
}
```

## 5. Email Notification (Optional)

```ts filename="src/actions/todos.ts"
import { courier } from '@towerjs/tower/courier'

export async function addTodo(formData: FormData) {
  // ... save todo

  await courier.email.send({
    to: session.user.email,
    subject: 'Todo Created',
    html: `<p>Created: ${title}</p>`,
  })
}
```

Requires Courier config in `tower.config.ts`:

```ts
courier: { email: { provider: 'console', from: 'noreply@example.com' } }
```
