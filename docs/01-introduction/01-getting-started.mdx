### Project Structure

A freshly scaffolded Tower project includes:

```
my-app/
├── src/
│   ├── proxy.ts                        # Auth middleware
│   ├── app/
│   │   ├── actions.ts                  # Server actions
│   │   ├── api/auth/[...all]/route.ts  # Auth API routes
│   │   ├── layout.tsx
│   │   └── page.tsx
│   └── vault/
│       ├── migrations/                 # Database migrations
│       └── seeds/                      # Seed data
├── tower.config.ts
├── next.config.ts
├── package.json
└── tsconfig.json
```

## Manual Setup

Add Tower to an existing Next.js project:

### Install

<CodeTabs>
  <CodeTab title="pnpm">
  ```bash
  pnpm add towerjs @towerjs/gatehouse
  ```
  </CodeTab>
  <CodeTab title="npm">
  ```bash
  npm install towerjs @towerjs/gatehouse
  ```
  </CodeTab>
  <CodeTab title="yarn">
  ```bash
  yarn add towerjs @towerjs/gatehouse
  ```
  </CodeTab>
  <CodeTab title="bun">
  ```bash
  bun add towerjs @towerjs/gatehouse
  ```
  </CodeTab>
</CodeTabs>

> **Edge runtime only** — `@towerjs/edge` is only needed if you deploy to an edge runtime (Vercel Edge, Cloudflare Workers). Most apps run on the Node.js runtime and can skip it. If you do use an edge runtime, install it and wrap `next.config.ts`:
>
> ```bash
> pnpm add -D @towerjs/edge
> ```
>
> ```tsx filename="next.config.ts"
> import { withTowerEdge } from '@towerjs/edge'
> export default withTowerEdge({})
> ```

### 1. Configure tower.config.ts

```tsx filename="tower.config.ts"
import { defineTower } from 'towerjs/blueprint'

export default defineTower({
  modules: {
    vault: {},
    courier: { email: { provider: 'console', from: 'noreply@example.com' } },
    gatehouse: {
      provider: 'better-auth',
      appName: 'My App',
      baseURL: 'http://localhost:3000',
      secret: process.env.BETTER_AUTH_SECRET,
      credentials: { enabled: true },
    },
  },
})
```

### 2. Set up auth middleware

```tsx filename="src/proxy.ts"
import { gatehouse } from '@towerjs/gatehouse'

const { handler } = gatehouse.proxy({
  public: ['/', '/sign-in', '/sign-up'],
  redirectIfAuthenticated: ['/sign-in', '/sign-up'],
  redirectTo: '/sign-in',
  redirectAfterSignIn: '/dashboard',
})

export const proxy = handler
export const config = { matcher: ['/((?!_next/static|favicon.ico|api/auth).*)'] }
```

### 3. Add auth route handler

```tsx filename="src/app/api/auth/[...all]/route.ts"
export { GET, POST } from 'towerjs/gatehouse/next'
```

### 4. Create server actions

```tsx filename="src/app/actions.ts"
'use server'

export {
  signIn,
  signUp,
  signOut,
} from 'towerjs/gatehouse/actions'
```

### 5. Run migrations

```bash
npx tower migrate
```

This creates all auth tables (users, sessions, accounts, organizations, etc.).

### 6. Use in server components

```tsx
import { gatehouse } from 'towerjs/gatehouse'

export default async function Dashboard() {
  const session = await gatehouse.getSession()
  if (!session) return <p>Not signed in</p>

  return <p>Welcome {session.user.name}</p>
}
```
