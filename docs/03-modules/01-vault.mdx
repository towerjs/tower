Vault wraps [Kysely](https://kysely.dev) with connection pooling, migrations, seeds, and both Node and Edge runtime support. It is built on PostgreSQL with a scoped provider boundary (Neon HTTP and standard `pg`) — see [Infrastructure](/docs/infrastructure) for the curated philosophy and escape hatch.

## Setup

```ts filename="tower.config.ts"
import { defineTower } from '@towerjs/tower/blueprint'
import { vault } from '@towerjs/vault'

export default defineTower({
  modules: [
    vault({
      connectionString: process.env.DATABASE_URL, // defaults to DATABASE_URL
      pool: { max: 20, idleTimeoutMillis: 30000 },
    }),
  ],
})
```

## Typing your schema

`vault` is a Kysely instance. Describe your tables once and every query is checked — table names, column names, and result shapes. Use `Generated<T>` for columns the database fills in (`id`, `created_at`, etc.) — optional on insert, present on select:

```ts filename="tower.config.ts"
import { defineTower } from '@towerjs/tower/blueprint'
import { type Generated, vault } from '@towerjs/vault'

type Database = {
  users: {
    id: Generated<string>
    name: string
    email: string
    created_at: Generated<Date>
  }
}

export default defineTower({
  modules: [vault<Database>({ connectionString: process.env.DATABASE_URL })],
})
```

In application code, bind the schema with a cast:

```ts filename="src/vault/db.ts"
import type { VaultModule } from '@towerjs/vault'
import { vault } from '@towerjs/vault'
import type { Generated } from '@towerjs/vault'

export type Database = {
  users: { id: Generated<string>; name: string; email: string; created_at: Generated<Date> }
  profiles: { id: Generated<string>; userId: string; bio: string }
}

export const db = vault as unknown as VaultModule<Database>
```

Import `db` wherever you query. The examples below repeat a trimmed-down version of that file so each one stands alone.

`Generated<T>` and `sql` are re-exported from `@towerjs/vault` so you don't need to add `kysely` directly.

## Querying

```ts
import type { VaultModule } from '@towerjs/vault'
import { vault } from '@towerjs/vault'

const db = vault as unknown as VaultModule<{ users: { id: string; name: string; email: string } }>
const userId = 'usr_123'

// Select
const users = await db.selectFrom('users').selectAll().where('email', '=', 'user@example.com').execute()

// Insert
await db.insertInto('users').values({ id: userId, name: 'Alice', email: 'alice@example.com' }).execute()

// Update
await db.updateTable('users').set({ name: 'Alice Smith' }).where('id', '=', userId).execute()

// Delete
await db.deleteFrom('users').where('id', '=', userId).execute()
```

## Transactions

```ts
import type { VaultModule } from '@towerjs/vault'
import { vault } from '@towerjs/vault'

const db = vault as unknown as VaultModule<{
  users: { id: string; name: string; email: string }
  profiles: { id: string; userId: string; bio: string }
}>

await db.transaction(async (trx) => {
  const user = await trx
    .insertInto('users')
    .values({ id: 'usr_123', name: 'Alice', email: 'alice@example.com' })
    .returningAll()
    .executeTakeFirstOrThrow()

  await trx.insertInto('profiles').values({ id: 'pro_123', userId: user.id, bio: 'Hello' }).execute()
})
```

## Migrations

Create migration files in `src/vault/migrations/`:

```ts filename="src/vault/migrations/0001_projects.ts"
import { type Vault } from '@towerjs/vault'

export async function up(db: Vault) {
  await db.schema
    .createTable('projects')
    .addColumn('id', 'uuid', (col) => col.primaryKey().defaultTo(db.fn('gen_random_uuid')))
    .addColumn('name', 'varchar(255)', (col) => col.notNull())
    .addColumn('created_at', 'timestamptz', (col) => col.defaultTo(db.fn('now')))
    .execute()
}

export async function down(db: Vault) {
  await db.schema.dropTable('projects').execute()
}
```

## Seeds

Create seed files in `src/vault/seeds/`:

```ts filename="src/vault/seeds/example.ts"
import type { Vault } from '@towerjs/vault'

export default async function seed(db: Vault<{ projects: { id: string; name: string } }>) {
  await db.insertInto('projects').values({ id: 'prj_123', name: 'My First Project' }).execute()
}
```

Run seeds:

```bash
npx tower seed
```

By default, migrations run before seeds. Skip migrations with `npx tower seed --skip-migrate`.

## Advanced: Using Kysely Functions

```ts
import type { VaultModule } from '@towerjs/vault'
import { vault } from '@towerjs/vault'

const db = vault as unknown as VaultModule<{ users: { id: string; name: string; email: string } }>

const result = await db.selectFrom('users').select(db.fn.countAll().as('count')).executeTakeFirst()
```

## Edge Runtime

On Edge (Vercel Edge, Cloudflare Workers), Vault uses the same Neon HTTP dialect (`kysely-neon`) as everywhere else — no connection pool is involved. Migrations and seeds are unavailable on Edge (filesystem-dependent).

```ts
import { vault } from '@towerjs/vault'

// vault.migrate()  // throws on Edge (filesystem unavailable)
// vault.seed()     // throws on Edge (filesystem unavailable)
```
