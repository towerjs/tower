---
title: "Gatehouse"
description: "Authentication module powered by Better Auth. Handles sign-in, sign-up, sessions, organizations, API keys, and more."
---

Gatehouse provides full-featured authentication with email/password, social login, passkeys, OTP, magic links, two-factor auth, organizations, and API keys.

## Setup

```tsx filename="tower.config.ts"
gatehouse: {
  provider: 'better-auth',
  appName: 'My App',
  baseURL: 'http://localhost:3000',
  secret: process.env.BETTER_AUTH_SECRET,

  // Enable auth methods
  credentials: { enabled: true },
  emailVerification: { sendOnSignUp: true, autoSignInAfterVerification: true },
  social: { google: {}, github: {} },
  magicLinks: true,
  passkeys: true,
  twoFactor: true,
  organization: true,
  admin: true,
  apiKey: true,
}
```

Social provider credentials resolve from env vars automatically:
`GOOGLE_CLIENT_ID`, `GOOGLE_CLIENT_SECRET`, `GITHUB_CLIENT_ID`, etc.

## Using Gatehouse

### In Server Components

```tsx
import { gatehouse } from 'towerjs/gatehouse'

export default async function DashboardPage() {
  const session = await gatehouse.getSession()
  if (!session) return <p>Not signed in</p>

  const user = await gatehouse.requireUser()
  const sessions = await gatehouse.getUserSessions()
  const orgs = await gatehouse.getOrganizations()
}
```

### In Server Actions

```tsx
'use server'
import { action } from 'towerjs/gatehouse/next'

export const myAction = action(async (formData: FormData) => {
  // gatehouse context is available via AsyncLocalStorage
  const session = await gatehouse.getSession()
})
```

### On the Client

```tsx
import { gatehouseClient } from 'towerjs/gatehouse/client'

async function handleSignIn() {
  await gatehouseClient.signIn.email({ email, password })
}
```

## Auth Methods API

### Sessions

```tsx
gatehouse.getSession()                           // Promise<Session | null>
gatehouse.requireUser()                          // Promise<GatehouseUser>
gatehouse.getUserSessions()                      // Promise<GatehouseSession[]>
```

### Sign In / Sign Up

```tsx
gatehouse.signIn.email({ email, password })       // Promise<Session>
gatehouse.signIn.emailOtp({ email, code })        // Promise<Session>
gatehouse.signIn.magicLink({ email })              // Promise<void>
gatehouse.signIn.phone({ phoneNumber, code })      // Promise<Session>
gatehouse.signIn.social({ provider })               // Promise<Session>
gatehouse.signUp.email({ name, email, password })  // Promise<Session>
gatehouse.signOut()                                // Promise<void>
```

### Account Management

```tsx
gatehouse.account.update({ name, image })          // Promise<GatehouseUser>
gatehouse.account.delete()                         // Promise<void>
gatehouse.account.setPassword(newPassword)          // Promise<void>
gatehouse.account.changeEmail(newEmail)             // Promise<void>
```

### Password

```tsx
gatehouse.password.change({ currentPassword, newPassword })  // Promise<void>
gatehouse.password.forgot({ email })                          // Promise<void>
gatehouse.password.reset({ newPassword, token })               // Promise<void>
gatehouse.password.confirm({ password })                       // Promise<boolean>
```

### Two-Factor Auth

```tsx
gatehouse.totp.enable(password)                    // Promise<TwoFactorInfo>
gatehouse.totp.disable(password)                   // Promise<void>
gatehouse.totp.verify(code)                        // Promise<TwoFactorVerifyResult>
gatehouse.backupCodes.generate(password)            // Promise<string[]>
gatehouse.backupCodes.verify(code)                  // Promise<TwoFactorVerifyResult>
```

### Organizations

```tsx
gatehouse.organizations.create({ name, slug })       // Promise<Organization>
gatehouse.organizations.list()                       // Promise<Organization[]>
gatehouse.organizations.get(orgId)                  // Promise<Organization | null>
gatehouse.organizations.setActive(orgId)             // Promise<void>
gatehouse.organizations.update(orgId, params)        // Promise<Organization>
gatehouse.organizations.delete(orgId)                // Promise<void>

gatehouse.organizations.members.list(orgId)          // Promise<OrganizationMember[]>
gatehouse.organizations.members.add(orgId, userId, role?)
gatehouse.organizations.members.update(orgId, memberId, role)
gatehouse.organizations.members.remove(orgId, memberId)

gatehouse.organizations.invitations.create(orgId, params)
gatehouse.organizations.invitations.accept(invitationId)
gatehouse.organizations.invitations.reject(invitationId)
```

### API Keys

```tsx
gatehouse.apiKeys.create(params)                    // Promise<ApiKeyInfo>
gatehouse.apiKeys.list(userId)                      // Promise<{ keys, total? }>
gatehouse.apiKeys.update(id, params)                 // Promise<ApiKeyInfo>
gatehouse.apiKeys.delete(id)                         // Promise<void>
gatehouse.apiKeys.verify(params)                     // Promise<ApiKeyInfo | null>
```

### Admin

```tsx
gatehouse.admin.createUser(params)                   // Promise<GatehouseUser>
gatehouse.admin.listUsers(options?)                   // Promise<{ users, total? }>
gatehouse.admin.banUser(userId)                       // Promise<void>
gatehouse.admin.impersonateUser(userId)               // Promise<...>
```

## Proxy Middleware

The proxy handles session cookie validation, redirects, and route protection:

```tsx filename="src/proxy.ts"
import { gatehouse } from '@towerjs/gatehouse'

const { handler } = gatehouse.proxy({
  public: ['/', '/sign-in'],           // unauthenticated routes
  redirectIfAuthenticated: ['/sign-in'],// redirect to /dashboard if already logged in
  redirectTo: '/sign-in',               // redirect here if not authenticated
  redirectAfterSignIn: '/dashboard',     // redirect here after sign in
})

export const proxy = handler
export const config = { matcher: ['/((?!_next/static|favicon.ico|api/auth).*)'] }
```

## Courier Auto-Wiring

When both gatehouse and courier are configured, gatehouse automatically uses courier for auth emails:
- Password reset emails
- Email verification
- Magic links
- Email OTP

No callback configuration needed.
