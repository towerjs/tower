Courier delivers notifications through email, SMS, and push channels with pluggable providers.

## Setup

```ts
import { courier } from '@towerjs/courier'

courier({
  email: {
    provider: 'console', // logs to stdout (no credentials)
    from: 'noreply@example.com',
  },
})
```

## Email Providers

### Console (Development)

```ts
import { courier } from '@towerjs/courier'

courier({
  email: { provider: 'console', from: 'noreply@example.com' },
})
```

Logs all email content to stdout. No credentials needed.

### Resend

```ts
import { courier } from '@towerjs/courier'

courier({
  email: { provider: 'resend', from: 'noreply@example.com' },
})
```

Requires `RESEND_API_KEY` environment variable.

### SMTP

```ts
import { courier } from '@towerjs/courier'

courier({
  email: {
    provider: 'smtp',
    from: 'noreply@example.com',
    host: 'smtp.example.com',
    port: 587,
    user: 'username',
    password: 'password',
  },
})
```

### SES

```ts
import { courier } from '@towerjs/courier'

courier({
  email: {
    provider: 'ses',
    from: 'noreply@example.com',
    region: 'us-east-1',
    accessKeyId: process.env.AWS_ACCESS_KEY_ID,
    secretAccessKey: process.env.AWS_SECRET_ACCESS_KEY,
  },
})
```

## SMS

### Twilio

```ts
import { courier } from '@towerjs/courier'

courier({
  sms: {
    provider: 'twilio',
    accountSid: process.env.TWILIO_ACCOUNT_SID,
    authToken: process.env.TWILIO_AUTH_TOKEN,
    messagingServiceSid: 'MG...',
  },
})
```

## Push

### Web Push

```ts
import { courier } from '@towerjs/courier'

courier({
  push: {
    provider: 'web-push',
    // VAPID keys generated by web-push library
  },
})
```

## Sending

```ts
import { courier } from '@towerjs/courier'

// Email
await courier.email.send({
  to: 'user@example.com',
  subject: 'Welcome!',
  html: '<h1>Hello!</h1>',
  text: 'Hello!',
})

// SMS
await courier.sms.send({
  to: '+1234567890',
  body: 'Your code is 123456',
})

// Push
await courier.push.send({
  subscription: pushSubscription,
  payload: { title: 'New message', body: 'You have a notification' },
})
```

## React Email Templates

```ts
import { WelcomeEmail } from '@/emails/welcome'

await courier.email.send({
  to: 'user@example.com',
  subject: 'Welcome!',
  react: WelcomeEmail({ name: 'Alice' }),
})
```

## Types

```ts
interface EmailSendParams {
  to: string | string[]
  subject: string
  from?: string
  cc?: EmailAddress
  bcc?: EmailAddress
  replyTo?: string
  text?: string
  html?: string
  react?: ReactElement
  headers?: Record<string, string>
  attachments?: EmailAttachment[]
}

interface SmsSendParams {
  to: string
  body: string
  from?: string
}

interface PushSendParams {
  subscription: PushSubscription
  payload?: string | Record<string, unknown>
  title?: string
  body?: string
  data?: Record<string, unknown>
  ttl?: number
  urgency?: 'very-low' | 'low' | 'normal' | 'high'
  topic?: string
}
```
